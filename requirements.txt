# credscheck-nxc - wymagane i opcjonalne narzędzia zewnętrzne
#
# WYMAGANE (bez tego skrypt nie wystartuje):
#   sudo apt install -y nmap
#   pipx install git+https://github.com/Pennyw0rth/NetExec
#
# OPCJONALNE - klienci do interaktywnego łączenia po znalezieniu poświadczeń
# (bez nich skrypt nadal działa, ale pokaże komunikat "narzędzie nie znalezione"
# zamiast otworzyć sesję):
#   sudo apt install -y smbclient          # SMB
#   sudo apt install -y sshpass            # SSH (auto-login hasłem)
#   sudo apt install -y lftp               # FTP
#   sudo apt install -y freerdp2-x11       # RDP (xfreerdp)
#   sudo apt install -y python3-impacket   # MSSQL, WMI
#   sudo apt install -y nfs-common         # NFS (showmount)
#   sudo apt install -y tigervnc-viewer    # VNC
#   gem install evil-winrm                 # WinRM (wymaga Ruby)
